⚙️ Supabase 配置
📖 如何获取配置
1. 访问 supabase.com 注册并创建项目（免费）
2. 在项目 Settings → API 中复制 Project URL 和 anon public key
3. 在 SQL Editor 中执行建表语句（见下方）
4. 在 Settings → Database 中开启 Realtime（homework 和 checkin 表）

建表 SQL：
CREATE TABLE IF NOT EXISTS homework ( id bigserial primary key, "user" text not null, subject text not null, content text not null, attachments jsonb default '[]', created_at timestamptz default now() ); CREATE TABLE IF NOT EXISTS checkin ( id bigserial primary key, "user" text not null, subject text not null, duration int default 0, start_time text, end_time text, date text, attachments jsonb default '[]', images jsonb default '[]', homework_ids jsonb default '[]', created_at timestamptz default now() ); CREATE TABLE IF NOT EXISTS music_plan ( id bigserial primary key, weekday int not null, play_time text not null, end_time text, resource_ids jsonb default '[]', enabled boolean default true, created_at timestamptz default now() ); ALTER TABLE music_plan ADD COLUMN IF NOT EXISTS end_time text; ALTER TABLE homework ENABLE ROW LEVEL SECURITY; ALTER TABLE checkin ENABLE ROW LEVEL SECURITY; ALTER TABLE music_plan ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "open" ON homework; DROP POLICY IF EXISTS "open" ON checkin; DROP POLICY IF EXISTS "open" ON music_plan; CREATE POLICY "open" ON homework FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "open" ON checkin FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "open" ON music_plan FOR ALL USING (true) WITH CHECK (true);
